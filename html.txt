  <script>
    const STORAGE_KEY = "sbu_episode_creation_system_v1";
    const SERVER_ENABLED = location.protocol.startsWith("http");
    const stages = [
      { id: "discover", name: "Discover", short: "Question" },
      { id: "story", name: "Story", short: "Real story" },
      { id: "truth", name: "Truth", short: "Scripture" },
      { id: "build", name: "Build", short: "Episode" },
      { id: "brotherhood", name: "Brotherhood", short: "Community" }
    ];

    const storyQuestions = [
      "Tell me about a time when you personally experienced this struggle.",
      "Where were you, and what was happening around you?",
      "What was the specific moment where the weight of it really hit you?",
      "What were you feeling right there?",
      "What were you afraid might happen?",
      "What did you tell yourself in that moment?",
      "What changed, even if it changed slowly?",
      "When you look back now, what do you understand that you did not understand then?",
      "How does that story connect to the central question of this episode?"
    ];

    const state = {
      currentEpisodeId: "",
      currentStage: "discover",
      episodes: []
    };
    let systemStatus = { ok: false, aiAvailable: false, model: "", storage: "" };

    function createEpisode() {
      const id = "ep-" + Date.now();
      return {
        id,
        updatedAt: new Date().toISOString(),
        meta: {
          episodeNumber: "",
          title: "Untitled SBU Episode",
          titleIdeas: [],
          targetLength: "30",
          audience: "men who feel like they are struggling alone"
        },
        discover: {
          centralQuestion: "",
          uncomfortableTruth: "",
          whyItMatters: "",
          listenerRecognition: "",
          listenerOutcome: "",
          emotionalJourney: "",
          coreTheme: ""
        },
        story: {
          questionIndex: 0,
          answers: [],
          draft: "",
          approved: false
        },
        truth: {
          useScripture: "undecided",
          scripture: "",
          translation: "NLT",
          plainMeaning: "",
          context: "",
          crossReferences: "",
          application: "",
          whyItFits: "",
          notes: ""
        },
        build: {
          hostName: "Robert",
          includeShowNotes: true,
          includeTimestamps: true,
          includeTitleIdeas: true,
          script: "",
          quality: []
        },
        brotherhood: {
          content: ""
        }
      };
    }

    function load() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (raw) {
          const parsed = JSON.parse(raw);
          state.episodes = Array.isArray(parsed.episodes) ? parsed.episodes : [];
          state.currentEpisodeId = parsed.currentEpisodeId || "";
          state.currentStage = parsed.currentStage || "discover";
        }
      } catch {
        state.episodes = [];
      }
      if (!state.episodes.length) {
        const ep = createEpisode();
        state.episodes.push(ep);
        state.currentEpisodeId = ep.id;
      }
      if (!getEpisode()) state.currentEpisodeId = state.episodes[0].id;
    }

    function save(showMessage = false) {
      const ep = getEpisode();
      if (ep) ep.updatedAt = new Date().toISOString();
      localStorage.setItem(STORAGE_KEY, JSON.stringify({
        currentEpisodeId: state.currentEpisodeId,
        currentStage: state.currentStage,
        episodes: state.episodes
      }));
      renderShell();
      if (SERVER_ENABLED) syncWorkspace(false);
      if (showMessage) toast("Workspace saved.");
    }

    function getEpisode() {
      return state.episodes.find(ep => ep.id === state.currentEpisodeId);
    }

    function escapeHtml(value) {
      return String(value ?? "")
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;");
    }

    function nl2br(value) {
      return escapeHtml(value).replace(/\n/g, "<br>");
    }

    function field(path, label, options = {}) {
      const ep = getEpisode();
      const value = path.split(".").reduce((acc, key) => acc?.[key], ep) ?? "";
      const id = path.replaceAll(".", "-");
      const type = options.type || "textarea";
      const hint = options.hint ? `<p class="hint">${escapeHtml(options.hint)}</p>` : "";
      if (type === "input") {
        return `<div class="field ${options.full ? "full" : ""}">
          <label for="${id}">${escapeHtml(label)}</label>
          <input id="${id}" data-path="${path}" value="${escapeHtml(value)}" placeholder="${escapeHtml(options.placeholder || "")}">
          ${hint}
        </div>`;
      }
      if (type === "select") {
        const opts = options.options.map(opt => `<option value="${escapeHtml(opt.value)}" ${String(value) === String(opt.value) ? "selected" : ""}>${escapeHtml(opt.label)}</option>`).join("");
        return `<div class="field ${options.full ? "full" : ""}">
          <label for="${id}">${escapeHtml(label)}</label>
          <select id="${id}" data-path="${path}">${opts}</select>
          ${hint}
        </div>`;
      }
      return `<div class="field ${options.full ? "full" : ""}">
        <label for="${id}">${escapeHtml(label)}</label>
        <textarea id="${id}" data-path="${path}" class="${options.long ? "long-textarea" : "short-textarea"}" placeholder="${escapeHtml(options.placeholder || "")}">${escapeHtml(value)}</textarea>
        ${hint}
      </div>`;
    }

    function setPath(path, value) {
      const parts = path.split(".");
      let target = getEpisode();
      for (let i = 0; i < parts.length - 1; i++) target = target[parts[i]];
      target[parts[parts.length - 1]] = value;
    }

    function completion(ep, stageId) {
      if (stageId === "discover") {
        return Boolean(ep.discover.centralQuestion && ep.discover.listenerOutcome);
      }
      if (stageId === "story") return Boolean(ep.story.draft || ep.story.answers.length >= 4);
      if (stageId === "truth") return ep.truth.useScripture !== "undecided";
      if (stageId === "build") return Boolean(ep.build.script);
      if (stageId === "brotherhood") return Boolean(ep.brotherhood.content);
      return false;
    }

    function renderShell() {
      const ep = getEpisode();
      document.getElementById("episodeSelect").innerHTML = state.episodes
        .map(item => `<option value="${item.id}" ${item.id === ep.id ? "selected" : ""}>${escapeHtml(item.meta.title || "Untitled SBU Episode")}</option>`)
        .join("");

      document.getElementById("episodeList").innerHTML = state.episodes
        .slice()
        .sort((a, b) => new Date(b.updatedAt) - new Date(a.updatedAt))
        .map(item => `<button class="episode-item ${item.id === ep.id ? "active" : ""}" data-episode-id="${item.id}">
          <strong>${escapeHtml(item.meta.title || "Untitled SBU Episode")}</strong>
          <span>${escapeHtml(item.discover.centralQuestion || "No central question yet")}</span>
        </button>`)
        .join("");

      document.getElementById("stageNav").innerHTML = stages
        .map((stage, index) => `<button class="stage-btn ${state.currentStage === stage.id ? "active" : ""}" data-stage="${stage.id}">
          <span class="stage-num">${index + 1}</span>
          <span><strong>${stage.name}</strong><br><span class="stage-status">${stage.short}</span></span>
          <span class="stage-status">${completion(ep, stage.id) ? "Done" : "Open"}</span>
        </button>`)
        .join("");

      const status = document.getElementById("systemStatus");
      if (status) {
        if (!SERVER_ENABLED) status.textContent = "Local file mode";
        else if (!systemStatus.ok) status.textContent = "Server not connected";
        else status.textContent = systemStatus.aiAvailable ? `AI on: ${systemStatus.model}` : "Local engines on";
      }
    }

    function render() {
      renderShell();
      const panel = document.getElementById("stagePanel");
      const stage = state.currentStage;
      if (stage === "discover") panel.innerHTML = renderDiscover();
      if (stage === "story") panel.innerHTML = renderStory();
      if (stage === "truth") panel.innerHTML = renderTruth();
      if (stage === "build") panel.innerHTML = renderBuild();
      if (stage === "brotherhood") panel.innerHTML = renderBrotherhood();
      bindPanelEvents();
    }

    function stageHeader(num, title, body) {
      return `<div class="stage-header">
        <div>
          <div class="eyebrow">Stage ${num}</div>
          <h2>${escapeHtml(title)}</h2>
          <p>${escapeHtml(body)}</p>
        </div>
        <button class="btn small" data-action="save">Save</button>
      </div>`;
    }

    function controls(prev, next) {
      return `<div class="stage-controls">
        <div class="left-controls">${prev ? `<button class="btn" data-goto="${prev}">Back</button>` : ""}</div>
        <div class="right-controls">
          <button class="btn" data-action="save">Save</button>
          ${next ? `<button class="btn primary" data-goto="${next}">Next</button>` : ""}
        </div>
      </div>`;
    }

    function renderDiscover() {
      return `${stageHeader(1, "Discover", "Find what this episode is really about before writing anything. Keep it simple, honest, and specific.")}
        <div class="notice">Start with the question. The app will use these answers later to keep the episode from becoming generic advice.</div>
        <div class="grid-2">
          ${field("meta.episodeNumber", "Episode number", { type: "input", placeholder: "Example: 12" })}
          ${field("meta.title", "Working episode title", { type: "input", placeholder: "Example: When You Feel Like You Failed" })}
          <div class="field full">
            <label>Episode title ideas</label>
            <div class="pill-row">
              <button class="btn small" data-action="generateTitleIdeas">Generate SBU Title Ideas</button>
              ${(getEpisode().meta.titleIdeas || []).map(title => `<button class="btn small" data-title-choice="${escapeHtml(title)}">${escapeHtml(title)}</button>`).join("")}
            </div>
            <p class="hint">Pick one to make it the working title, or keep editing the title manually.</p>
          </div>
          ${field("discover.centralQuestion", "What is the ONE question this episode needs to answer?", { full: true, placeholder: "Example: What do you do when you feel like you have failed everyone who counted on you?" })}
          ${field("discover.uncomfortableTruth", "What is the uncomfortable truth underneath that question?", { full: true })}
          ${field("discover.whyItMatters", "Why does this matter to struggling men?", { full: true })}
          ${field("discover.listenerRecognition", "What should the listener recognize about himself?", {})}
          ${field("discover.listenerOutcome", "What should the listener understand differently by the end?", {})}
          ${field("discover.emotionalJourney", "What emotional journey should the episode take him through?", { full: true })}
          ${field("discover.coreTheme", "Core theme", { type: "input", placeholder: "Example: shame, isolation, failure, forgiveness, hope" })}
        </div>
        ${controls("", "story")}`;
    }

    function renderStory() {
      const ep = getEpisode();
      const q = storyQuestions[Math.min(ep.story.questionIndex, storyQuestions.length - 1)];

    function bindPanelEvents() {
      document.querySelectorAll("[data-path]").forEach(el => {

    function bindGlobalEvents() {
      document.body.addEventListener("click", event => {
        const target = event.target.closest("button");
        if (!target) return;
        const stage = target.dataset.stage;
        const goto = target.dataset.goto;
        const action = target.dataset.action;
        const episodeId = target.dataset.episodeId;
        const titleChoice = target.dataset.titleChoice;
        if (stage) {
          state.currentStage = stage;
          save();
          render();
        }
        if (goto) {
          state.currentStage = goto;
          save();
          render();
          window.scrollTo({ top: 0, behavior: "smooth" });
        }
        if (episodeId) {
          state.currentEpisodeId = episodeId;
          save();
          render();
        }
        if (titleChoice) {
          getEpisode().meta.title = titleChoice;
          save(true);
          render();
        }
        if (action) handleAction(action);
      });

      document.getElementById("episodeSelect").addEventListener("change", event => {
        state.currentEpisodeId = event.target.value;
        save();
        render();
      });
      document.getElementById("saveBtn").addEventListener("click", () => save(true));
      document.getElementById("syncBtn").addEventListener("click", () => syncWorkspace(true));
      document.getElementById("newEpisodeBtn").addEventListener("click", () => {
        const ep = createEpisode();
        state.episodes.push(ep);
        state.currentEpisodeId = ep.id;
        state.currentStage = "discover";
        save(true);
        render();
      });
      document.getElementById("deleteEpisodeBtn").addEventListener("click", () => {
        if (state.episodes.length === 1) {
          toast("Keep at least one episode in the workspace.");
          return;
        }
        const ep = getEpisode();
        const ok = confirm(`Delete "${ep.meta.title || "Untitled SBU Episode"}"?`);
        if (!ok) return;
        state.episodes = state.episodes.filter(item => item.id !== ep.id);
        state.currentEpisodeId = state.episodes[0].id;
        save(true);
        render();
      });
      document.getElementById("exportJsonBtn").addEventListener("click", exportWorkspace);
    }

    function handleAction(action) {
      if (action === "save") save(true);
      if (action === "answerStory") answerStory();
      if (action === "generateTitleIdeas") generateTitleIdeas();
      if (action === "generateStoryDraft") generateStoryDraft();
      if (action === "engineStory") runEngine("story");
      if (action === "resetStoryQuestion") resetStoryQuestion();
      if (action === "suggestScripture") suggestScripture();
      if (action === "engineTruth") runEngine("truth");
      if (action === "markNoScripture") markNoScripture();
      if (action === "generateEpisode") generateEpisode();
      if (action === "engineEpisode") runEngine("episode");
      if (action === "qualityCheck") qualityCheck(true);
      if (action === "engineQuality") runEngine("quality");
      if (action === "copyScript") copyText(getEpisode().build.script, "Script copied.");
      if (action === "downloadTxt") downloadFile(fileBase() + "-episode.txt", getEpisode().build.script, "text/plain");
      if (action === "downloadDoc") downloadDoc();
      if (action === "generateBrotherhood") generateBrotherhood();
      if (action === "engineBrotherhood") runEngine("brotherhood");
      if (action === "copyBrotherhood") copyText(getEpisode().brotherhood.content, "Brotherhood content copied.");
      if (action === "downloadBrotherhood") downloadFile(fileBase() + "-brotherhood.txt", getEpisode().brotherhood.content, "text/plain");
    }

    function answerStory() {
      const ep = getEpisode();
      const input = document.getElementById("storyAnswer");
      const answer = input.value.trim();
      if (!answer) {
        toast("Answer the current story question first.");
        return;
      }
      ep.story.answers.push({
        question: storyQuestions[ep.story.questionIndex],
        answer
      });
      ep.story.questionIndex += 1;
      save(true);
      render();
    }

    function generateTitleIdeas() {
      const ep = getEpisode();
      const theme = ep.discover.coreTheme || "the struggle";
      const question = ep.discover.centralQuestion || "the question no man wants to ask out loud";
      const cleanQuestion = question.replace(/[?!.]+$/g, "");
      ep.meta.titleIdeas = [
        `When ${capitalize(theme)} Gets Heavy`,
        `The Question Under the Struggle`,
        `You Do Not Have to Carry This Alone`,
        `What If Strength Is Telling the Truth`,
        `Still Standing`,
        cleanQuestion.length < 64 ? cleanQuestion : `The Truth Beneath ${capitalize(theme)}`
      ];
      if (!ep.meta.title || ep.meta.title === "Untitled SBU Episode") ep.meta.title = ep.meta.titleIdeas[0];
      save(true);
      render();
    }

    function capitalize(value) {
      const text = String(value || "").trim();
      return text ? text.charAt(0).toUpperCase() + text.slice(1) : text;
    }

    function generateStoryDraft() {
      const ep = getEpisode();
      if (!ep.story.answers.length) {
        toast("Answer a few story questions first.");
        return;
      }
      const parts = ep.story.answers.map(item => item.answer.trim()).filter(Boolean);
      ep.story.draft = [
        "Robert's real story draft:",
        "",
        parts.join("\n\n"),
        "",
        "Connection to the episode:",
        ep.story.answers.at(-1)?.answer || "Add the connection after the interview is complete.",
        "",
        "Integrity note: This draft only uses details Robert entered in the Story Finder. Do not add facts, motives, dialogue, locations, or outcomes unless Robert supplies them."
      ].join("\n");
      save(true);
      render();
    }

    function resetStoryQuestion() {
      const ok = confirm("Restart the story interview? Existing answers will stay, but the next question will return to the beginning.");
      if (!ok) return;
      getEpisode().story.questionIndex = 0;
      save(true);
      render();
    }

    function suggestScripture() {
      const ep = getEpisode();
      const theme = `${ep.discover.coreTheme} ${ep.discover.centralQuestion}`.toLowerCase();
      let suggestion = "Psalm 34:18";
      let reason = "This passage speaks gently to men who feel brokenhearted or crushed without pretending pain disappears quickly.";
      if (theme.includes("shame") || theme.includes("failure")) {
        suggestion = "Romans 8:1";
        reason = "This passage fits when the episode is dealing with shame, condemnation, and the fear that failure gets the final word.";
      } else if (theme.includes("alone") || theme.includes("isolation")) {
        suggestion = "Ecclesiastes 4:9-10";
        reason = "This passage fits the brotherhood theme because it names the danger of falling alone and the strength of having someone beside you.";
      } else if (theme.includes("fear")) {
        suggestion = "Isaiah 41:10";
        reason = "This passage can support a fear episode if it is handled as God being present with us, not as pressure to pretend we are fearless.";
      } else if (theme.includes("forgive")) {
        suggestion = "Colossians 3:13";
        reason = "This passage fits a forgiveness episode if the conversation stays honest about pain, repair, and responsibility.";
      }
      ep.truth.useScripture = "explore";
      ep.truth.scripture = ep.truth.scripture || suggestion;
      ep.truth.whyItFits = ep.truth.whyItFits || reason;
      ep.truth.plainMeaning = ep.truth.plainMeaning || "Say this in plain language Robert would actually use. Keep it grounded, honest, and connected to the story.";
      ep.truth.notes = ep.truth.notes || "Do not force this passage. If the episode does not naturally need Scripture, skip it.";
      save(true);
      render();
    }

    async function checkServerStatus() {
      if (!SERVER_ENABLED) {
        renderShell();
        return;
      }
      try {
        const res = await fetch("/api/status");
        systemStatus = await res.json();
      } catch {
        systemStatus = { ok: false, aiAvailable: false };
      }
      renderShell();
    }

    async function syncWorkspace(showMessage) {
      if (!SERVER_ENABLED) {
        if (showMessage) toast("Open through the local server to sync to disk.");
        return;
      }
      try {
        const payload = {
          currentEpisodeId: state.currentEpisodeId,
          currentStage: state.currentStage,
          episodes: state.episodes
        };
        await fetch("/api/workspace", {
          method: "POST",
          headers: { "content-type": "application/json" },
          body: JSON.stringify({ workspace: payload })
        });
        if (showMessage) toast("Workspace synced to disk.");
      } catch {
        if (showMessage) toast("Disk sync failed. Browser save still works.");
      }
    }

    async function runEngine(engine) {
      if (!SERVER_ENABLED) {
        toast("Start the local server to use server engines.");
        return;
      }
      const ep = getEpisode();
      toast("Running SBU engine...");
      try {
        const res = await fetch(`/api/engine/${engine}`, {
          method: "POST",
          headers: { "content-type": "application/json" },
          body: JSON.stringify({ episode: ep })
        });
        const result = await res.json();
        if (!res.ok) throw new Error(result.error || "Engine failed");
        applyEngineResult(engine, result.text || "");
        save(true);
        render();
        toast(result.mode === "ai" ? "AI engine complete." : "Local engine complete.");
      } catch (error) {
        toast(error.message || "Engine failed.");
      }
    }

    function applyEngineResult(engine, text) {
      const ep = getEpisode();
      if (engine === "story") {
        ep.story.draft = text;
      } else if (engine === "truth") {
        ep.truth.notes = text;
        if (ep.truth.useScripture === "undecided") ep.truth.useScripture = "explore";
      } else if (engine === "episode") {
        ep.build.script = text;
        qualityCheck(false);
      } else if (engine === "brotherhood") {
        ep.brotherhood.content = text;
      } else if (engine === "quality") {
        ep.build.quality = text.split("\n").filter(Boolean).map(line => ({
          label: line,
          pass: /^pass/i.test(line),
          fix: line
        }));
      }
    }

    function qualityCheck(showToast) {
      const ep = getEpisode();
      const script = ep.build.script || "";
      const truthUsed = ep.truth.useScripture;
      ep.build.quality = [
        check("Real human story is present", Boolean(ep.story.draft.trim()), "Add or approve a real story in Story Finder. Do not invent one."),
        check("Central question is clear", Boolean(ep.discover.centralQuestion.trim()), "Add the one question this episode answers."),
        check("Story connects to the episode", Boolean(ep.discover.whyItMatters.trim()), "Explain why the story matters to struggling men."),
        check("Scripture is not forced", truthUsed === "skip" || Boolean(ep.truth.whyItFits.trim()), "Explain why the passage fits, or mark Scripture as skipped."),
        check("Episode circles back to the opening", script.includes("11. CIRCLE BACK") && !script.includes("[CIRCLE BACK BLOCKED]"), "Add the real story so the ending can return to it."),
        check("Practical next step is included", script.includes("9. PRACTICAL NEXT STEPS"), "Generate or add practical next steps."),
        check("Brotherhood challenge is included", script.includes("10. BROTHERHOOD CHALLENGE"), "Generate or add a brotherhood challenge."),
        check("Hopeful close is included", script.includes("12. HOPEFUL CLOSE"), "Generate or add the hopeful close.")
      ];
      save(false);
      if (showToast) {
        toast("Quality check complete.");
        render();
      }
    }

    function check(label, pass, fix) {
      return { label, pass, fix };
    }

    async function copyText(text, message) {
      if (!text) {
        toast("Nothing to copy yet.");
        return;
      }
      if (navigator.clipboard?.writeText) {
        try {
          await navigator.clipboard.writeText(text);
          toast(message);
          return;
        } catch {}
      }
      const helper = document.createElement("textarea");
      helper.value = text;
      helper.style.position = "fixed";
      helper.style.left = "-9999px";
      document.body.appendChild(helper);
      helper.focus();
      helper.select();
      document.execCommand("copy");
      helper.remove();
      toast(message);
    }

    function fileBase() {
      const ep = getEpisode();
      return (ep.meta.title || "sbu-episode").toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "") || "sbu-episode";
    }

    function downloadFile(filename, text, type) {
      if (!text) {
        toast("Nothing to download yet.");
        return;
      }
      const blob = new Blob([text], { type });
      const a = document.createElement("a");
      a.href = URL.createObjectURL(blob);
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      a.remove();
      URL.revokeObjectURL(a.href);
    }

    function downloadDoc() {
      const script = getEpisode().build.script;
      if (!script) {
        toast("Generate the script first.");
        return;
      }
      const html = `<html><head><meta charset="utf-8"></head><body><pre style="white-space:pre-wrap;font-family:Calibri,Arial,sans-serif;">${escapeHtml(script)}</pre></body></html>`;
      downloadFile(fileBase() + "-episode.doc", "\ufeff" + html, "application/msword");
    }

    function exportWorkspace() {
      downloadFile("sbu-episode-workspace.json", JSON.stringify(getEpisode(), null, 2), "application/json");
    }

    function toast(message) {
      const el = document.getElementById("toast");
      el.textContent = message;
      el.classList.add("show");
      clearTimeout(window.__toastTimer);
      window.__toastTimer = setTimeout(() => el.classList.remove("show"), 1800);
    }

    load();
    bindGlobalEvents();
    render();
    checkServerStatus();
    setInterval(() => save(false), 20000);
  </script>
